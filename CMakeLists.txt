cmake_minimum_required(VERSION 3.20)
project(peek_host C)
set(CMAKE_C_STANDARD 11)
include(cmake/dragon_assets.cmake)
add_library(dragon src/dragon.c src/dragon_render.c)
add_dependencies(dragon peek_dragon_assets)
target_include_directories(dragon PUBLIC src PRIVATE "${PEEK_GENERATED_DIR}")
target_compile_options(dragon PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
target_link_libraries(dragon PUBLIC m)
add_library(peek src/peek.c src/replay.c)
target_include_directories(peek PUBLIC src)
target_compile_options(peek PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
target_link_libraries(peek PUBLIC m)
add_executable(test_detector tests/test_detector.c)
target_link_libraries(test_detector PRIVATE peek)
target_compile_options(test_detector PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_executable(test_device_adapter tests/test_device_adapter.c)
target_include_directories(test_device_adapter PRIVATE tests/stubs)
target_compile_options(test_device_adapter PRIVATE -UNDEBUG -Wall -Wextra -Werror)
target_link_libraries(test_device_adapter PRIVATE peek)
add_executable(test_ui_lifetime tests/test_ui_lifetime.c src/peek_ui.c)
target_include_directories(test_ui_lifetime PRIVATE tests/stubs)
target_compile_options(test_ui_lifetime PRIVATE -UNDEBUG -Wall -Wextra -Werror)
target_link_libraries(test_ui_lifetime PRIVATE peek dragon)
add_executable(peek_replay tools/replay.c)
target_link_libraries(peek_replay PRIVATE peek)
enable_testing()
add_test(NAME detector COMMAND test_detector)
add_test(NAME device_adapter COMMAND test_device_adapter)
add_test(NAME ui_lifetime COMMAND test_ui_lifetime)

add_executable(test_event_aging tests/test_event_aging.c)
target_compile_options(test_event_aging PRIVATE -UNDEBUG -Wall -Wextra -Werror)
target_link_libraries(test_event_aging PRIVATE peek)
add_test(NAME event_aging COMMAND test_event_aging)
add_executable(peek_replay_until tools/replay_until.c)
target_link_libraries(peek_replay_until PRIVATE peek)

add_executable(test_dragon tests/test_dragon.c)
target_link_libraries(test_dragon PRIVATE peek dragon)
target_compile_options(test_dragon PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_test(NAME dragon_behavior COMMAND test_dragon)
add_executable(test_dragon_render tests/test_dragon_render.c)
target_link_libraries(test_dragon_render PRIVATE dragon)
target_compile_options(test_dragon_render PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_test(NAME dragon_pixels COMMAND test_dragon_render)
add_executable(dragon_snapshot tools/dragon_snapshot.c)
target_link_libraries(dragon_snapshot PRIVATE dragon)
add_executable(test_dragon_timing tests/test_dragon_timing.c)
target_link_libraries(test_dragon_timing PRIVATE peek dragon)
target_compile_options(test_dragon_timing PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_test(NAME dragon_receive_timing COMMAND test_dragon_timing)
add_test(NAME asset_build_readonly COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_asset_build.py")
