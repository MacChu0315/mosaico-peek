cmake_minimum_required(VERSION 3.20)
project(peek_host C)
set(CMAKE_C_STANDARD 11)
add_library(peek src/peek.c src/replay.c)
target_include_directories(peek PUBLIC src)
target_compile_options(peek PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
target_link_libraries(peek PUBLIC m)
add_executable(test_detector tests/test_detector.c)
target_link_libraries(test_detector PRIVATE peek)
target_compile_options(test_detector PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_executable(test_device_adapter tests/test_device_adapter.c)
target_include_directories(test_device_adapter PRIVATE tests/stubs)
target_compile_options(test_device_adapter PRIVATE -UNDEBUG -Wall -Wextra -Werror)
target_link_libraries(test_device_adapter PRIVATE peek)
add_executable(test_ui_lifetime tests/test_ui_lifetime.c src/peek_ui.c)
target_include_directories(test_ui_lifetime PRIVATE tests/stubs)
target_compile_options(test_ui_lifetime PRIVATE -UNDEBUG -Wall -Wextra -Werror)
target_link_libraries(test_ui_lifetime PRIVATE peek)
add_executable(peek_replay tools/replay.c)
target_link_libraries(peek_replay PRIVATE peek)
enable_testing()
add_test(NAME detector COMMAND test_detector)
add_test(NAME device_adapter COMMAND test_device_adapter)
add_test(NAME ui_lifetime COMMAND test_ui_lifetime)
